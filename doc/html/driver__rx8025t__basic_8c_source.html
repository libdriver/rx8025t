280 return 1;

282 timeptr.tm_year = t.year - 1900;
283 timeptr.tm_mon = t.month - 1;
284 timeptr.tm_wday = t.week;
285 timeptr.tm_mday = t.date;
286 timeptr.tm_hour = t.hour;
287 timeptr.tm_min = t.minute;
